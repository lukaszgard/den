{{- /* The latest releases as cards, from CHANGELOG.md (mounted as assets/den/CHANGELOG.md): the version and date of the top entries, their bold lead line, and the paragraph that follows it. Each card links to its entry on the changelog page. @param {int} count How many releases to show. Default 3. @example {{< den/whats-new >}} */ -}} {{- $count := int (.Get "count" | default 3) -}} {{- $changelog := resources.Get "den/CHANGELOG.md" -}} {{- if not $changelog -}} {{- errorf "den/whats-new: den/CHANGELOG.md is not mounted (see module.mounts in hugo.yaml)" -}} {{- end -}} {{- $entries := findRESubmatch `(?m)^## \[([^\]]+)\] - (\d{4}-\d{2}-\d{2})\n\n\*\*([^*]+)\*\*([\s\S]*?)\n\n` $changelog.Content $count -}} {{- if not $entries -}} {{- errorf "den/whats-new: CHANGELOG.md has no '## [X.Y.Z] - date' entry opening with a bold line" -}} {{- end -}} {{- $page := site.GetPage "/docs/changelog" -}}
{{- range $entries }} {{- $version := index . 1 -}} {{- $date := index . 2 -}} {{- $lead := index . 3 -}} {{- $rest := index . 4 | replaceRE `\s+` " " | strings.TrimSpace -}}

{{ $version }} · {{ $date }}

{{ $lead | markdownify }}

{{ $rest | markdownify }}

 {{- end }}
